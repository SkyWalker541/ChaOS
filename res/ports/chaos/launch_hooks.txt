# ChaOS: PortMaster's own app on ChaOS. PortMaster's utils/pm_launch.txt
# sources this when it is started as `source pm_launch.txt chaos` (ChaOS's
# res/ports/run.sh --portmaster), and these replace its defaults: no sudo, no
# text console, no systemd, and ChaOS's own Python and SDL. run.sh copies this
# file into PortMaster/chaos/ before every start, as it does mod_ChaOS.txt,
# so a PortMaster update can't leave it out.

pm_hook_env() {
  export ESUDO=""
  export ESUDOKILL="-1"
  source "$controlfolder/control.txt"
  [ -f "${controlfolder}/mod_${CFW_NAME}.txt" ] && source "${controlfolder}/mod_${CFW_NAME}.txt"
  get_controls
}

pm_hook_gui_env() {
  # pySDL2 finds SDL2 and SDL2_gfx in ChaOS's Python first: SDL2 as ports
  # get it (run.sh copies /usr/lib/chaos-ports' there: it reads the pad and
  # keys without udev; the system's doesn't, and the app saw no button), and
  # SDL2_gfx built with the names pySDL2 asks for (the system's has them as
  # GFX_*); then the system's SDL2_image, _ttf and _mixer
  export PYSDL2_DLL_PATH="${PYTHONHOME:+$PYTHONHOME/lib/sdl:}/usr/lib"
  # The Brick has no udev, so SDL may not find the pad by itself: named
  # here, as the Brick Gamepad's own event device
  local ev
  ev=$(awk '/^N: Name="Brick Gamepad"/{p=1} p && /^H: Handlers=/{for(i=1;i<=NF;i++) if($i ~ /^event[0-9]+$/){print $i; exit}}' /proc/bus/input/devices 2> /dev/null)
  [ -n "$ev" ] && export SDL_JOYSTICK_DEVICE="/dev/input/$ev"
  # and what SDL makes of it, in the log: how many pads, and whether each
  # is a game controller (the app needs that to read its buttons)
  CHAOS_PAD_OK=$(PYTHONPATH="$controlfolder/exlibs" python3 - << 'PYEOF' 2>&1 | tee -a /dev/stderr | grep -c "controller$"
import os, ctypes, sdl2
sdl2.SDL_Init(sdl2.SDL_INIT_GAMECONTROLLER)
n = sdl2.SDL_NumJoysticks()
print("chaos pad: SDL from %s; SDL_JOYSTICK_DEVICE=%s, %d joystick(s)" % (getattr(getattr(sdl2.dll, "dll", None), "libfile", "?"), os.environ.get("SDL_JOYSTICK_DEVICE"), n))
for i in range(n):
    g = sdl2.SDL_JoystickGetDeviceGUID(i)
    b = (ctypes.c_char * 33)()
    sdl2.SDL_JoystickGetGUIDString(g, b, 33)
    print("chaos pad:", i, sdl2.SDL_JoystickNameForIndex(i), b.value,
          "controller" if sdl2.SDL_IsGameController(i) else "NOT a controller")
sdl2.SDL_Quit()
PYEOF
)
  export CHAOS_PAD_OK
}

pm_hook_autoinstall_dirs() {
  echo "$controlfolder/autoinstall"
  echo "/$directory/ports/autoinstall"
}

pm_hook_pre_launch() {
  # no text console to clear or write to (CUR_TTY is /dev/null)
  chmod -R +x . 2> /dev/null
  :
}

pm_hook_run_pugwash() {
  # If SDL found no game controller, the pad is turned into the keys the app
  # also reads (arrows, Z = A, X = B, Return = Start, Escape = Select...) by
  # PortMaster's own gptokeyb, for as long as the app runs
  local gk=""
  if [ "${CHAOS_PAD_OK:-0}" = 0 ] && [ -x "$controlfolder/gptokeyb" ]; then
    printf '%s\n' "up = up" "down = down" "left = left" "right = right" \
      "a = z" "b = x" "x = a" "y = s" "l1 = q" "r1 = e" "start = enter" "back = esc" \
      > /tmp/chaos-pugwash.gptk
    "$controlfolder/gptokeyb" -1 "python3" -c /tmp/chaos-pugwash.gptk &
    gk=$!
    echo "chaos pad: no controller in SDL, so the pad as keys (gptokeyb $gk)"
  fi
  # The pad by its labels for the app: ports get it by position (the Xbox
  # way, so the Brick's A, on the right, is SDL's B), and the app then had A
  # and B, X and Y the wrong way round. Its own copy, the app's alone.
  local pad="$CHAOS_PAD"
  pad=$(printf '%s\n' "$pad" | sed 's/a:b1,b:b0,x:b3,y:b2/a:b0,b:b1,x:b2,y:b3/')
  printf '%s\n' "$pad" > /tmp/chaos-pugwash-pad.txt
  # ChaOS's Python (run.sh put it first on PATH); pugwash is #!/usr/bin/env python3
  SDL_GAMECONTROLLERCONFIG="$pad" SDL_GAMECONTROLLERCONFIG_FILE=/tmp/chaos-pugwash-pad.txt \
    python3 ./pugwash $PORTMASTER_CMDS
  [ -n "$gk" ] && kill "$gk" 2> /dev/null
  :
}

pm_hook_post_launch() {
  :
}

pm_hook_frontend_refresh() {
  # ChaOS rescans the Ports shelf itself when PortMaster closes; a marker any
  # of PortMaster's refreshes leave is only cleared
  rm -f "$controlfolder/.emustation-refresh" "$controlfolder/.emulationstation-refresh" \
        "$controlfolder/.weston-refresh" "$controlfolder/.sway-refresh" \
        "$controlfolder/.batocera-es-refresh" "$controlfolder/.trimui-refresh" 2> /dev/null
  :
}
