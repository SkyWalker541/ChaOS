## Modular - ChaOS (github.com/SkyWalker541/ChaOS), the TrimUI Brick
#
# Sourced by ports after control.txt, as PortMaster's other mod_<CFW>.txt are:
#   [ -f "${controlfolder}/mod_${CFW_NAME}.txt" ] && source "${controlfolder}/mod_${CFW_NAME}.txt"
# CFW_NAME is ChaOS from OS_NAME in /etc/os-release. ChaOS puts this file back
# in PortMaster's folder before every port (ChaOS/ports/run.sh), so an update
# of PortMaster that leaves it out loses nothing.

# Nothing to clear: the launcher has let go of the screen, and there is no
# text console on it
CUR_TTY=/dev/null

# The Brick's screen, said outright: what PortMaster reads from the frame
# buffer can be its double-height virtual size
export DISPLAY_WIDTH=1024
export DISPLAY_HEIGHT=768
export ASPECT_X=4
export ASPECT_Y=3
export DISPLAY_ORIENTATION=0
export ANALOG_STICKS=0

# Godot 2, and 3 and later
GODOT2_OPTS="-r ${DISPLAY_WIDTH}x${DISPLAY_HEIGHT} -f"
GODOT_OPTS="--resolution ${DISPLAY_WIDTH}x${DISPLAY_HEIGHT} -f"

# The pad, for SDL and gptokeyb: "Brick Gamepad" (plastron's brick-pad), keys
# only. Laid out by position as an Xbox pad is, as PortMaster expects: the
# bottom face button (B on the Brick) is SDL's a, the right one (A) its b.
# CHAOS_PAD is set by run.sh, the same lines SDL_GAMECONTROLLERCONFIG has.
get_controls() {
  ANALOGSTICKS="0"
  LOWRES="N"
  DEVICE="19000000427269636b2047616d657000"
  param_device="brick"

  # The Brick's own lines only, as PortMaster's get_controls keeps only the
  # device's: ports put this file's text in SDL_GAMECONTROLLERCONFIG, which
  # every program they start inherits, and PortMaster's whole list (460 KB)
  # is over the kernel's limit for one variable - nothing could start.
  export SDL_GAMECONTROLLERCONFIG_FILE="/tmp/gamecontrollerdb.txt"
  printf '%s\n' "$CHAOS_PAD" > /tmp/gamecontrollerdb.txt

  SDLDBUSERFILE="${HOME}/.config/SDL-GameControllerDB/gamecontrollerdb.txt"
  sdl_controllerconfig="$(< "${SDL_GAMECONTROLLERCONFIG_FILE}")"
}

pm_platform_helper() {
  pm_message_close

  if [ -e "$PM_PIPE" ]; then
    PortMasterDialogExit
  fi

  # gptokeyb's keyboard, before the game starts: a port starts gptokeyb in
  # the background and calls this just before its game. The port SDL finds a
  # keyboard that comes later too (run.sh), but a game that reads its keys
  # from the first frame would lose them. Up to 2 s, for a new event node
  # beyond the ones there when the port began.
  if pgrep -f gptokeyb > /dev/null 2>&1; then
    i=0
    while [ $i -lt 40 ]; do
      n=$(ls /dev/input/event* 2> /dev/null | wc -l)
      [ "$n" -gt "${CHAOS_EV_BASE:-0}" ] && break
      usleep 50000
      i=$((i + 1))
    done
  fi
}
